import streamlit as st import pdfplumber import json import pandas as pd from pydantic import BaseModel, Field from typing import List, Optional from google import genai from google.genai import types # -------------------------------------------------------------------- # 1. Pydantic Schema: Enforce Comprehensive Fundamental Analysis # -------------------------------------------------------------------- class KeyRatio(BaseModel): metric_name: str = Field(description="Name of metric (e.g., ROE, Net Margin, Current Ratio, Debt-to-Equity)") value: str = Field(description="Reported or computed value with units") health_signal: str = Field(description="'Bullish', 'Neutral', or 'Bearish'") interpretation: str = Field(description="Brief assessment of what this number means") class AspectAnalysis(BaseModel): profitability_and_growth: str = Field(description="Revenue growth, gross/operating/net margins, ROIC, and earnings quality") solvency_and_liquidity: str = Field(description="Debt structure, interest coverage, working capital, cash burn, and bankruptcy risks") operational_efficiency: str = Field(description="Asset turnover, inventory management, operating leverage") governance_and_risks: str = Field(description="Management commentary tone, capital allocation, legal/regulatory exposures, off-balance sheet items") class FundamentalAnalysisReport(BaseModel): company_name: str reporting_period: str currency: str executive_summary: str key_metrics: List[KeyRatio] detailed_aspects: AspectAnalysis moat_and_competitive_position: str bull_case: List[str] bear_case: List[str] valuation_and_verdict: str = Field(description="Verdict: Undervalued, Fairly Valued, Overvalued, or Distressed with explicit rationale") # -------------------------------------------------------------------- # 2. PDF Parsing Utilities # -------------------------------------------------------------------- def extract_text_and_tables(uploaded_file, max_pages: int = 40) -> str: """Extracts raw text and converts tabular data into clean Markdown format.""" extracted_content = [] with pdfplumber.open(uploaded_file) as pdf: num_pages = min(len(pdf.pages), max_pages) for i in range(num_pages): page = pdf.pages[i] text = page.extract_text() or "" # Extract tables if present tables = page.extract_tables() table_md = "" for table in tables: if table and len(table) > 1: df = pd.DataFrame(table[1:], columns=table[0]) table_md += "\n" + df.dropna(how="all").to_markdown(index=False) + "\n" extracted_content.append(f"--- PAGE {i + 1} ---\n{text}\n{table_md}") return "\n".join(extracted_content) # -------------------------------------------------------------------- # 3. AI Agent Core Analysis # -------------------------------------------------------------------- def run_fundamental_agent(parsed_financial_text: str) -> FundamentalAnalysisReport: client = genai.Client() prompt = f""" You are an institutional Chief Investment Officer and Wall Street Equity Research Analyst. Conduct a rigorous, institutional-grade fundamental analysis based solely on the provided financial statements, notes, and disclosures. Audit all core aspects: 1. Profitability & DuPont breakdown (ROE, ROIC, margin compression/expansion). 2. Liquidity & Capital Structure (debt maturity wall, cash runway, leverage). 3. Operating Efficiency (working capital cycle, asset turnover). 4. Qualitative Red Flags (MD&A tone shift, footnotes, contingent liabilities). 5. Valuation & Margin of Safety. Financial Report Data: \"\"\"{parsed_financial_text}\"\"\" """ response = client.models.generate_content( model="gemini-2.5-flash", contents=prompt, config=types.GenerateContentConfig( response_mime_type="application/json", response_schema=FundamentalAnalysisReport, temperature=0.1, # Low temperature for strict financial accuracy ), ) return FundamentalAnalysisReport.model_validate_json(response.text) # -------------------------------------------------------------------- # 4. Streamlit UI # -------------------------------------------------------------------- st.set_page_config(page_title="Institutional AI Fundamental Analyst", layout="wide") st.title("📊 AI Fundamental Analysis Terminal") st.markdown("Upload a quarterly/annual report (`10-K`, `10-Q`, or local audited financial statements) for an institutional multi-aspect audit.") uploaded_file = st.file_uploader("Upload Financial Report (PDF)", type=["pdf"]) if uploaded_file is not None: if st.button("Run Fundamental Audit", type="primary"): with st.status("Processing report...", expanded=True) as status: st.write("Extracting statements, notes, and balance sheet tables...") raw_text = extract_text_and_tables(uploaded_file) st.write("Executing multi-aspect financial audit via Gemini...") try: report = run_fundamental_agent(raw_text) status.update(label="Analysis complete!", state="complete", expanded=False) except Exception as e: status.update(label="Analysis failed", state="error") st.error(f"Error during analysis: {str(e)}") st.stop() # Render Header col1, col2, col3 = st.columns(3) col1.metric("Entity", report.company_name) col2.metric("Reporting Period", report.reporting_period) col3.metric("Currency", report.currency) st.subheader("Executive Summary") st.write(report.executive_summary) # Render Key Ratios Table st.subheader("Key Financial Ratios & Signals") ratios_data = [ { "Metric": r.metric_name, "Value": r.value, "Signal": "🟢 " + r.health_signal if r.health_signal.lower() == "bullish" else ("🔴 " + r.health_signal if r.health_signal.lower() == "bearish" else "🟡 " + r.health_signal), "Analysis": r.interpretation } for r in report.key_metrics ] st.dataframe(pd.DataFrame(ratios_data), use_container_width=True) # Render Core Pillars st.subheader("Core Fundamental Pillars") tab1, tab2, tab3, tab4 = st.tabs([ "Profitability & Growth", "Solvency & Liquidity", "Operational Efficiency", "Governance & Hidden Risks" ]) with tab1: st.markdown(report.detailed_aspects.profitability_and_growth) with tab2: st.markdown(report.detailed_aspects.solvency_and_liquidity) with tab3: st.markdown(report.detailed_aspects.operational_efficiency) with tab4: st.markdown(report.detailed_aspects.governance_and_risks) # Competitive Moat & Thesis st.subheader("Moat & Thesis Breakdown") st.write(f"**Competitive Advantage:** {report.moat_and_competitive_position}") col_bull, col_bear = st.columns(2) with col_bull: st.success("### Bull Case") for item in report.bull_case: st.markdown(f"- {item}") with col_bear: st.error("### Bear Case & Tail Risks") for item in report.bear_case: st.markdown(f"- {item}") # Verdict st.subheader("Valuation Verdict") st.info(report.valuation_and_verdict)
