import json from playwright.sync_api import sync_playwright def get_live_csx_data(username, password): with sync_playwright() as p: # Launch browser instance browser = p.chromium.launch(headless=True) context = browser.new_context() page = context.new_page() # Listen to incoming network responses def handle_response(response): # Target internal APIs delivering market depth or prices if "market" in response.url or "orderbook" in response.url: try: data = response.json() print("Captured Live Orderbook Data:", json.dumps(data, indent=2)) except Exception: pass page.on("response", handle_response) # 1. Navigate to CSX portal login page.goto("https://i.csx.com.kh/") # 2. Input credentials (update selectors if CSX uses different IDs/names) page.fill("input[name='username']", username) page.fill("input[name='password']", password) page.click("button[type='submit']") # 3. Wait for the trading interface to render page.wait_for_selector(".trading-view", timeout=15000) # Close session browser.close() # Example execution call: # get_live_csx_data("pichar", "123456789")
