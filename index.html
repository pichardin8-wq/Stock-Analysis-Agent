import threading import time from fastapi import FastAPI from fastapi.responses import HTMLResponse from playwright.sync_api import sync_playwright # ========================================== # CSX CREDENTIALS (Consider using env variables) # ========================================== CSX_USER = "" CSX_PASS = "" # ========================================== market_cache = { "status": "Initializing... fetching live data.", "last_updated": None, "stocks": [] } def run_scraper_thread(): """Background worker that continuously refreshes stock data.""" global market_cache while True: try: print("[*] Background Worker: Fetching CSX data...") with sync_playwright() as p: browser = p.chromium.launch(headless=True) page = browser.new_page() page.goto("https://i.csx.com.kh/market-summary", wait_until="domcontentloaded") page.wait_for_timeout(2000) # Handle login if prompted username_field = page.locator('input[type="text"], input[name="username"], input[name="email"], input[id*="user"]') if username_field.count() > 0: username_field.first.fill(CSX_USER) page.locator('input[type="password"]').first.fill(CSX_PASS) submit_btn = page.locator('button[type="submit"], button:has-text("Login"), button:has-text("Log In")') submit_btn.first.click() page.wait_for_load_state("networkidle") page.wait_for_timeout(3000) # Extract table rows rows = page.eval_on_selector_all("table tbody tr", """ rows => rows.map(r => { const cells = Array.from(r.querySelectorAll('td, th')).map(c => c.innerText.trim()); return cells.filter(c => c.length > 0); }).filter(r => r.length > 0) """) if rows: market_cache = { "status": "Live", "last_updated": time.strftime("%Y-%m-%d %H:%M:%S"), "total_tickers": len(rows), "stocks": rows } print(f"[+] Updated {len(rows)} tickers at {market_cache['last_updated']}") else: market_cache["status"] = "Waiting for data..." print("[-] No rows retrieved. Will retry...") browser.close() except Exception as e: print(f"[!] Scraper Error: {e}") market_cache["status"] = f"Error: {e}" time.sleep(15) # Scrapes every 15 seconds # Start background sync threading.Thread(target=run_scraper_thread, daemon=True).start() app = FastAPI(title="CSX Live Market Dashboard") @app.get("/stocks") def get_stocks(): return market_cache @app.get("/", response_class=HTMLResponse) def index(): return """ CSX Live Market Board

Cambodia Securities Exchange (CSX)

Real-Time Market Watch
Checking... --
Connecting to feed...
"""
