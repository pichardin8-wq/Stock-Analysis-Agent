import os import threading import time from fastapi import FastAPI from fastapi.responses import HTMLResponse from playwright.sync_api import sync_playwright CSX_USER = os.getenv("CSX_USER", "pichar") CSX_PASS = os.getenv("CSX_PASS", "123456789") market_cache = { "status": "Connecting to exchange...", "last_updated": None, "stocks": [] } def run_scraper_thread(): """Continuously fetches stock table data from CSX.""" global market_cache while True: try: with sync_playwright() as p: browser = p.chromium.launch(headless=True) page = browser.new_page() page.goto("https://i.csx.com.kh/market-summary", wait_until="domcontentloaded") page.wait_for_timeout(2000) # Check and perform login if input fields are present user_field = page.locator('input[type="text"], input[name="username"], input[name="email"]') if user_field.count() > 0 and CSX_USER: user_field.first.fill(CSX_USER) page.locator('input[type="password"]').first.fill(CSX_PASS) submit_btn = page.locator('button[type="submit"], button:has-text("Login"), button:has-text("Log In")') submit_btn.first.click() page.wait_for_load_state("networkidle") page.wait_for_timeout(2000) # Extract market table entries rows = page.eval_on_selector_all("table tbody tr", """ rows => rows.map(r => { const cells = Array.from(r.querySelectorAll('td, th')).map(c => c.innerText.trim()); return cells.filter(c => c.length > 0); }).filter(r => r.length > 0) """) if rows: market_cache = { "status": "Live", "last_updated": time.strftime("%H:%M:%S ICT"), "stocks": rows } else: market_cache["status"] = "No open market data" browser.close() except Exception as err: market_cache["status"] = f"Scraper Alert: {err}" time.sleep(15) # Initialize background scraper threading.Thread(target=run_scraper_thread, daemon=True).start() app = FastAPI(title="CSX Live Market Terminal") @app.get("/stocks") def get_stocks(): return market_cache DASHBOARD_HTML = ''' CSX Live Terminal
